export const s = story('CD19', 'P15391-2')
export const record = s.record
export const transcripts = cd19Transcripts().transcripts
export const canonical = transcripts.find(t => t.tags.includes('MANE_Select'))
export const skip = skipping(canonical, 2)
export const skipped = skipsExon(transcripts, canonical, 2)
export const coding = transcripts.filter(
  t =>
    t.type === 'protein_coding' &&
    !t.tags.includes('cds_start_NF') &&
    !t.tags.includes('cds_end_NF'),
)
export const others = transcripts.filter(t => !coding.includes(t))
export const otherTypes = [
  ...new Set(others.map(t => t.type.replace(/_/g, ' '))),
].join(', ')
export const [, first, second] = [0, ...record.exons.junctions]
export const residues = [Math.ceil((first + 1) / 3), Math.ceil(second / 3)]
export const domain = record.features.find(f => f.type === 'Domain')
export const exonLength = skip.skipped[1] - skip.skipped[0] + 1
export const noteText =
  codingNote(record.coding, record.gene) === ''
    ? 'so the page has no sentence about other transcripts'
    : 'and the page gives the number'
export const skippedText =
  skipped.length === 0
    ? `none of the ${transcripts.length} joins exon 1 directly to exon 3`
    : `${skipped.map(t => t.name).join(', ')} join exon 1 directly to exon 3`
export const extra = record.isoforms[0].length - record.length

[Sotillo et al. (2015)](https://doi.org/10.1158/2159-8290.CD-15-1020) write
that CD19, "expressed on most B-cell acute lymphoblastic leukemias (B-ALL), can
be targeted with chimeric antigen receptor-armed T cells (CART-19), but
relapses with epitope loss occur in 10% to 20% of pediatric responders". Besides deletions of the _CD19_ locus and mutations in exon 2 in
some relapse samples, they "discovered alternatively spliced CD19 mRNA species,
including one lacking exon 2", and showed that exon 2 skipping "allows
expression of the N-terminally truncated CD19 variant, which fails to trigger
killing by CART-19".

## Exon 2 on the canonical protein

In CD19's MANE Select transcript, {canonical.id.replace(/\.\d+$/, '')}, exon 2
is {exonLength} bases long and holds coding nucleotides{' '}
{span(first + 1, second)}, residues {span(...residues)} of the canonical
protein, with the first and last of those codons split across the junctions.
Those residues cover most of the first Ig-like domain
({span(domain.start, domain.end)}). {exonLength} is{' '}

exons after it. The Exons row on the{' '}

residues it encodes.

Where GENCODE 49 lists complete protein-coding transcripts that UniProt maps to
no isoform, a gene page gives their number, since those have no page here. For CD19,

et al. report, which is outside both UniProt and GENCODE 49.
